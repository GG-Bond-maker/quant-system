+def _seed_all_empty(present, codes: list[str]) -> None:
+    """当日与上一交易日均无数据（present 保持空）——全体走 suspended 分支。"""
+    for c in codes:
+        present[TRADE_DAY].pop(_sym(c), None)
+        present[PREV_DAY].pop(_sym(c), None)
