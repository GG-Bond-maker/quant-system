```python
"""Task 1: hfq 结算口径——除权日持仓市值不缩水、涨跌停基准为除权参考价。"""
from datetime import date
import pandas as pd

def _run_split_scenario(open_prices, close_prices):
    """同一 10送10 场景：d1 买入，d1/d2 收盘估值，d2 为除权日。
    返回 (d1 市值, d2 市值)。喂哪种口径的价，broker 就按哪种口径估值。"""
    from app.backtest.broker import Broker
    uni = pd.DataFrame({
        "open": open_prices, "close": close_prices, "volume": [1e6] * 4,
        "limit_up":   [c * 1.1 for c in close_prices],
        "limit_down": [c * 0.9 for c in close_prices],
        "is_halted": [False] * 4,
    }, index=["600001.SH"] * 4)
    b = Broker(init_cash=1_000_000)
    d1, d2 = date(2024, 1, 2), date(2024, 1, 3)
    b.buy(d1, "600001.SH", 500_000.0, uni.iloc[1])
    b.mark_to_market(d1, uni.iloc[[1]])
    mv_before = b.total_equity
    b.mark_to_market(d2, uni.iloc[[2]])
    return mv_before, b.total_equity

def test_split_event_no_fake_loss():
    """对照规格：raw 口径除权缺口被记为 ~-50% 亏损（缺陷机制存档），
    hfq 口径无虚假亏损。本测试修复前后都应通过（Broker 零逻辑改动，
    它固化的是语义而非本次修复的红灯）——真正的 TDD 红灯在 Step 2。"""
    mv_b, mv_a = _run_split_scenario(
        [88.0, 99.0, 50.0, 51.0], [90.0, 100.0, 50.0, 51.0])       # raw
    assert mv_a < mv_b * 0.6            # raw：-50% 假亏损被记为真亏损
    mv_b, mv_a = _run_split_scenario(
        [88.0, 99.0, 100.0, 101.0], [90.0, 100.0, 100.0, 102.0])   # hfq
    assert abs(mv_a - mv_b) < 1.0       # hfq：无虚假亏损
```
